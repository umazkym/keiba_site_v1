# 06. データ処理パイプライン

## 6.1 パイプライン概要

データ処理パイプラインは、netkeiba.comからレース情報を取得し、AI予測を生成してデータベースに保存する一連の処理です。

### 6.1.1 パイプラインの実行タイミング

**Render Cronジョブで自動実行** (1日5回):

| 時刻(JST) | 時刻(UTC) | モード | 処理内容 |
|----------|----------|--------|---------|
| 11:00 | 02:00 | PREDICTIONS_ONLY | 翌日の予測生成 |
| 05:00 | 20:00 | RESULTS_ONLY | 前日の結果取得 |
| 07:00 | 22:00 | SNS_POST (morning) | 的中報告投稿 |
| 12:00 | 03:00 | SNS_POST (afternoon) | レース一覧投稿 |
| 20:00 | 11:00 | SNS_POST (evening) | 翌日注目馬投稿 |

#### get_race_list_jra()

**役割**: JRAのレース一覧を取得

**データソース**: `https://race.netkeiba.com/api/api_get_jra_race_list.html?kaisai_date={YYYYMMDD}`

**実装**:
```python
def get_race_list_jra(date: datetime.date) -> List[Dict]:
    date_str = date.strftime('%Y%m%d')
    url = f"https://race.netkeiba.com/api/api_get_jra_race_list.html?kaisai_date={date_str}"

    response = requests.get(url, headers=get_random_headers())
    time.sleep(random.uniform(2.5, 5.0))  # Ban対策

    if response.status_code != 200:
        return []

    data = response.json()
    races = []

    for venue in data:
        for race in venue['RaceList']:
            races.append({
                'race_id': race['RaceID'],
                'venue_name': venue['RaceCourse'],
                'race_number': race['RaceNum'],
                'race_name': race['RaceName'],
                'race_type': 'JRA'
            })

    return races
```

class ReturnProcessor:
    def parse_returns(self, soup: BeautifulSoup) -> List[Dict]:
        returns = []

        # 単勝・複勝
        tansho_rows = soup.select('.Tansho tr')
        for row in tansho_rows:
            returns.append({
                'bet_type': '単勝',
                'number_1': int(row.select_one('.Num').text),
                'payout': int(row.select_one('.Payout').text.replace(',', '')),
                'popularity': int(row.select_one('.Popular').text)
            })

        # 3連単
        sanrentan_rows = soup.select('.Sanrentan tr')
        for row in sanrentan_rows:
            numbers = row.select_one('.Num').text.split('-')
            returns.append({
                'bet_type': '3連単',
                'number_1': int(numbers[0]),
                'number_2': int(numbers[1]),
                'number_3': int(numbers[2]),
                'payout': int(row.select_one('.Payout').text.replace(',', '')),
                'popularity': int(row.select_one('.Popular').text)
            })

        return returns

#### save_race_and_horses()

**役割**: レース情報と出走馬をDB保存

**実装**:
```python
from sqlalchemy.orm import Session
from backend.database.models import Race, Horse, Jockey, Trainer

def save_race_and_horses(db: Session, race_data: Dict, horses_data: List[Dict]):
    # Raceレコード作成
    race = Race(
        id=race_data['race_id'],
        race_date=race_data['race_date'],
        venue_name=race_data['venue_name'],
        race_number=race_data['race_number'],
        race_name=race_data.get('race_name'),
        race_type=race_data['race_type'],
        course_type=race_data.get('course_type'),
        distance=race_data.get('distance'),
        weather=race_data.get('weather'),
        ground_condition=race_data.get('ground_condition'),
        total_horses=len(horses_data)
    )

    # 既存レコードがあれば更新
    existing = db.query(Race).filter(Race.id == race.id).first()
    if existing:
        for key, value in race.__dict__.items():
            if key != '_sa_instance_state':
                setattr(existing, key, value)
    else:
        db.add(race)

    # 馬・騎手・調教師のマスタ更新
    for horse_data in horses_data:
        # Horse
        horse = db.query(Horse).filter(Horse.id == horse_data['horse_id']).first()
        if not horse:
            horse = Horse(
                id=horse_data['horse_id'],
                name=horse_data['horse_name'],
                sex=horse_data.get('sex'),
                age=horse_data.get('age')
            )
            db.add(horse)

        # Jockey
        if horse_data.get('jockey_id'):
            jockey = db.query(Jockey).filter(Jockey.id == horse_data['jockey_id']).first()
            if not jockey:
                jockey = Jockey(
                    id=horse_data['jockey_id'],
                    name=horse_data['jockey_name']
                )
                db.add(jockey)

        # Trainer (同様の処理)

    db.commit()
```

## 6.6 バッチ処理とメモリ管理

### 6.6.1 バッチサイズ設定

```python
import os
import gc

# 環境に応じたバッチサイズ
BATCH_SIZE = int(os.getenv('BATCH_SIZE', '15'))  # Render: 15, ローカル: 50

def process_races_in_batches(races: List[Dict]):
    for i in range(0, len(races), BATCH_SIZE):
        batch = races[i:i + BATCH_SIZE]

        logger.info(f"Processing batch {i // BATCH_SIZE + 1}")

        for race in batch:
            process_single_race(race)

        # ガベージコレクション
        gc.collect()

        # 小休止
        time.sleep(1)
```

## 6.7 エラーハンドリング

### 6.7.1 リトライロジック

```python
from tenacity import retry, stop_after_attempt, wait_exponential

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=4, max=10)
)
def fetch_with_retry(url: str) -> str:
    response = requests.get(url, timeout=30)
    response.raise_for_status()
    return response.text
```
